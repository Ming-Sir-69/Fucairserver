cmake_minimum_required(VERSION 3.20)
project(FucairserverMiracast LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# The C++/WinRT upstream MinGW tests also avoid GCC's devirtualization bug.
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
if(NOT WIN32 OR NOT MINGW)
    message(FATAL_ERROR "This host currently targets the project's MinGW Windows toolchain")
endif()
set(FUCAIRSERVER_CPPWINRT_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/miracast-reference/research/projection" CACHE PATH "Generated C++/WinRT projection")
if(NOT EXISTS "${FUCAIRSERVER_CPPWINRT_INCLUDE}/winrt/Windows.Media.Miracast.h")
    message(FATAL_ERROR "C++/WinRT projection is absent; run scripts/Build-Miracast.ps1")
endif()
add_executable(Fucairserver.Miracast WIN32 main.cpp Fucairserver.rc)
target_include_directories(Fucairserver.Miracast SYSTEM PRIVATE "${FUCAIRSERVER_CPPWINRT_INCLUDE}")
target_compile_definitions(Fucairserver.Miracast PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(Fucairserver.Miracast PRIVATE -mcx16 -fno-devirtualize -finput-charset=UTF-8)
target_link_options(Fucairserver.Miracast PRIVATE -municode -static -static-libgcc -static-libstdc++)
target_link_libraries(Fucairserver.Miracast PRIVATE runtimeobject synchronization ole32 oleaut32 user32 gdi32 shell32)
